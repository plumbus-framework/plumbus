# Testing UI Generators

The UI package uses Vitest for generator tests. The package also has a browser-test script that can run through `vitest.config.browser.ts` when browser coverage is configured.

```bash
pnpm --filter @plumbus/ui test
pnpm --filter @plumbus/ui test:browser
```

Generator tests should verify behavior at the source-string boundary. Most generators return strings or `{ path, content }` objects, so tests should check exported names, generated paths, important source fragments, omitted optional features, and parseable JSON where applicable. Higher-level compile tests can be added for generated files when the surrounding test harness supports them.

## Test files

```text
src/generators/__tests__/
  client-generator.test.ts
  auth-generator.test.ts
  form-generator.test.ts
  nextjs-template.test.ts
  translation-generator.test.ts
```

## Testing strategy

Use `toContain()` for stable generated fragments such as function names, import paths, route paths, and file headers. Use `not.toContain()` to verify omitted features, such as tenant helpers when `multiTenant` is false or capability pages when calling `generateNextjsTemplate`. Parse JSON outputs such as `package.json` and `tsconfig.json` before asserting their fields. For multi-file generators, map `GeneratedFile[]` to paths and assert the returned file set.

## Test a capability-based client generator

```ts
import { describe, expect, it } from "vitest";
import type { CapabilityContract } from "@plumbus/core";
import { generateTypedClient } from "../client-generator.js";

const mockCap = {
  name: "getUser",
  kind: "query",
  domain: "users",
  description: "Get a user by ID",
  input: {} as any,
  output: {} as any,
  effects: { data: [], events: [], external: [], ai: false },
  handler: async () => ({}),
  exposeAs: ["api"],
} as unknown as CapabilityContract;

describe("generateTypedClient", () => {
  it("generates a GET client for query capabilities", () => {
    const code = generateTypedClient(mockCap);

    expect(code).toContain("export async function getUser");
    expect(code).toContain('method: "GET"');
    expect(code).toContain("/api/users/get-user");
  });

  it("includes JSDoc when configured", () => {
    const code = generateTypedClient(mockCap, { includeJsDoc: true });

    expect(code).toContain("/** Get a user by ID */");
  });

  it("uses a custom base URL", () => {
    const code = generateTypedClient(mockCap, {
      baseUrl: "https://api.example.com",
    });

    expect(code).toContain("https://api.example.com/api/users/get-user");
  });
});
```

## Test full client and hook modules

```ts
import { describe, expect, it } from "vitest";
import { generateClientModule, generateHooksModule } from "../client-generator.js";

describe("client and hook modules", () => {
  it("generates a client module from capabilities and flows", () => {
    const code = generateClientModule([mockCap], [], { includeJsDoc: true });

    expect(code).toContain("// Auto-generated by @plumbus/ui");
    expect(code).toContain("export interface PlumbusApiError");
    expect(code).toContain("export async function getUser");
  });

  it("generates React hooks from capabilities", () => {
    const code = generateHooksModule([mockCap], { toastImport: "sonner" });

    expect(code).toContain('import { useState, useEffect } from "react"');
    expect(code).toContain('import { toast } from "sonner"');
    expect(code).toContain("export function useGetUser");
  });
});
```

`generateClientModule` and `generateHooksModule` omit capabilities that are not `exposeAs: ['api']`. Include that field on fixtures when asserting generated fetch or hook names.

## Test auth generation

```ts
import { describe, expect, it } from "vitest";
import { generateAuthModule } from "../auth-generator.js";

describe("generateAuthModule", () => {
  it("includes auth types, functions, hooks, and route guard", () => {
    const code = generateAuthModule({ provider: "jwt" });

    expect(code).toContain("interface AuthUser");
    expect(code).toContain("function login");
    expect(code).toContain("function useAuth");
    expect(code).toContain("function RouteGuard");
  });

  it("includes tenant context when multiTenant is enabled", () => {
    const code = generateAuthModule({ provider: "jwt", multiTenant: true });

    expect(code).toContain("function useTenant");
  });

  it("excludes tenant context by default", () => {
    const code = generateAuthModule({ provider: "jwt" });

    expect(code).not.toContain("function useTenant");
  });
});
```

## Test form hint extraction

```ts
import { describe, expect, it } from "vitest";
import { z } from "zod";
import type { CapabilityContract } from "@plumbus/core";
import { extractFieldHint, extractFormHints } from "../form-generator.js";

describe("extractFieldHint", () => {
  it("extracts string validation", () => {
    const hint = extractFieldHint("name", z.string().min(1).max(100));

    expect(hint.fieldType).toBe("text");
    expect(hint.required).toBe(true);
    expect(hint.validation.minLength).toBe(1);
    expect(hint.validation.maxLength).toBe(100);
  });

  it("extracts enum options", () => {
    const hint = extractFieldHint("role", z.enum(["admin", "user", "guest"]));

    expect(hint.fieldType).toBe("select");
    expect(hint.options).toEqual(["admin", "user", "guest"]);
  });
});

describe("extractFormHints", () => {
  it("extracts fields from capability.input", () => {
    const cap = {
      name: "createUser",
      kind: "action",
      domain: "users",
      input: z.object({
        name: z.string(),
        email: z.string().email(),
        role: z.enum(["admin", "user"]),
      }),
    } as unknown as CapabilityContract;

    const hints = extractFormHints(cap);

    expect(hints.capabilityName).toBe("createUser");
    expect(hints.fields).toHaveLength(3);
    expect(hints.fields[1]!.validation.pattern).toBe("email");
  });
});
```

## Test Next.js template generation

```ts
import { describe, expect, it } from "vitest";
import {
  generateCapabilityPage,
  generateNextjsTemplate,
  generatePackageJson,
} from "../nextjs-template.js";

describe("generateNextjsTemplate", () => {
  it("generates valid package.json with minimal dependency shape", () => {
    const file = generatePackageJson({ appName: "My App" });
    const parsed = JSON.parse(file.content);

    expect(parsed.name).toBe("my-app");
    expect(parsed.scripts.dev).toBe("next dev");
    expect(parsed.dependencies).toEqual({});
    expect(parsed.devDependencies).toEqual({});
  });

  it("includes auth files when auth is enabled", () => {
    const files = generateNextjsTemplate({ appName: "Test", auth: true });
    const paths = files.map((file) => file.path);

    expect(paths).toContain("components/AuthProvider.tsx");
    expect(paths).toContain("app/login/page.tsx");
    expect(paths).toContain("app/signup/page.tsx");
  });

  it("does not generate capability pages as part of the template", () => {
    const cap = { name: "getUser", kind: "query", domain: "users" } as any;
    const files = generateNextjsTemplate({ appName: "Test" }, [cap]);
    const paths = files.map((file) => file.path);

    expect(paths).not.toContain("app/get-user/page.tsx");
  });
});

describe("generateCapabilityPage", () => {
  it("generates one page file for a capability", () => {
    const cap = { name: "getUser", kind: "query", domain: "users" } as any;
    const file = generateCapabilityPage(cap);

    expect(file.path).toBe("app/get-user/page.tsx");
    expect(file.content).toContain("useGetUser");
  });
});
```

## Test translation generation

```ts
import { describe, expect, it } from "vitest";
import type { TranslationDefinition } from "@plumbus/core";
import { generateTranslationModule } from "../translation-generator.js";

const definitions: TranslationDefinition[] = [
  {
    name: "common",
    defaultLocale: "en",
    locales: ["en", "he"],
    messages: {
      en: { "nav.home": "Home" },
      he: { "nav.home": "בית" },
    },
  },
];

describe("generateTranslationModule", () => {
  it("returns no files without definitions", () => {
    expect(generateTranslationModule([])).toEqual([]);
  });

  it("generates i18n files from definitions", () => {
    const files = generateTranslationModule(definitions);
    const paths = files.map((file) => file.path);

    expect(paths).toContain("i18n/messages.ts");
    expect(paths).toContain("i18n/config.ts");
    expect(paths).toContain("i18n/translated-text.ts");
    expect(paths).toContain("i18n/request.ts");
    expect(paths).toContain("i18n/provider.tsx");
    expect(paths).toContain("i18n/index.ts");
  });

  it("generates split locale bundles when enabled", () => {
    const files = generateTranslationModule(definitions, { splitLocaleBundles: true });
    const paths = files.map((file) => file.path);

    expect(paths).toContain("i18n/locales/en.ts");
    expect(paths).toContain("i18n/locales/he.ts");
  });
});
```
